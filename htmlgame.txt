    <script>
        let tg = window.Telegram.WebApp;
        tg.expand();

        let energy = 100;
        let credits = 500;

        function updateStats() {
            document.getElementById('energy').innerText = energy;
            document.getElementById('credits').innerText = credits;
        }

        function nextStep(option) {
            let storyDiv = document.getElementById('story');
            let choicesDiv = document.getElementById('choices');

            if (option === 1) {
                energy -= 20;
                credits += 100;
                storyDiv.innerText = "Шлюз взломан грубой силой! Система подняла тревогу, но вы успели выкачать 100 кредитов со счетов корпорации.";
            } else {
                energy -= 10;
                credits += 250;
                storyDiv.innerText = "Тихий взлом прошел успешно. Инсайдер поделился ключами доступа к хранилищу. Получено 250 кредитов!";
            }
            updateStats();

            choicesDiv.innerHTML = `<button class="btn" onclick="resetGame()">Продолжить проникновение...</button>`;
        }

        function resetGame() {
            let storyDiv = document.getElementById('story');
            let choicesDiv = document.getElementById('choices');

            storyDiv.innerText = "Вы на втором уровне безопасности. Корпоративные дроны патрулируют периметр. Ваши действия?";
            choicesDiv.innerHTML = `
                <button class="btn" onclick="nextStep(1)">1. Проскользнуть мимо патруля</button>
                <button class="btn" onclick="nextStep(2)">2. Перегрузить сеть дронов</button>
            `;
        }
    </script>
